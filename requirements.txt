# Dependensi mesin biometrik (verifikasi KTP + wajah).
#
# ============================================================================
# WAJIB DIPASANG DI PYTHON 3.12 — BUKAN DI PYTHON SISTEM MESIN DEV.
#
# Produksi (VPS) memakai Python 3.12.3 dan kombinasi di bawah ini sudah
# terbukti berjalan di sana. Mesin dev Windows memakai Python 3.14 yang belum
# punya wheel TensorFlow, jadi butuh lingkungan terpisah:
#
#   py -3.12 -m venv .venv-ml
#   .venv-ml\Scripts\pip install -r requirements.txt
#   # lalu di .env:
#   PYTHON_BIN=D:\Tolongin\backend\.venv-ml\Scripts\python.exe
#
# PythonWorkerService membaca PYTHON_BIN untuk menjalankan worker biometrik.
# ============================================================================
#
# RIWAYAT dua kesalahan yang pernah membuat berkas ini tidak bisa dipakai —
# keduanya gagal dengan cara yang menyalahkan pengguna, bukan sistem:
#
# 1. tensorflow sempat tidak tercantum sama sekali, padahal DeepFace
#    membutuhkannya untuk membangun Facenet. Seluruh verifikasi wajah gagal
#    dengan "No module named 'tensorflow'", dan pengguna diberi tahu bahwa
#    wajahnya tidak cocok dengan KTP.
#
# 2. Versi sempat dipatok ke tensorflow-cpu 2.15.1 bersama opencv 4.13.
#    opencv 4.11+ mensyaratkan numpy>=2 sedangkan tensorflow-intel 2.15.1
#    mensyaratkan numpy<2, jadi `pip install -r requirements.txt` selalu
#    berakhir ResolutionImpossible dan lingkungan ML tidak pernah terpasang.
#
# Konflik numpy itu hilang sendiri pada TensorFlow 2.21, yang memang berjalan
# di atas numpy 2. Karena itu patokan mengikuti stack produksi, bukan
# sebaliknya — menurunkan TensorFlow demi mempertahankan pin lama justru akan
# merusak lingkungan yang sudah bekerja.
#
# PERINGATAN: mengganti versi TensorFlow atau opencv menggeser hasil pelurusan
# wajah, dan jendela aman antara pasangan sah dengan pasangan beda-orang hanya
# selebar ~0,075 (lihat src/ai/python/verify_face.py). Setelah menaikkan versi,
# ukur ulang jaraknya sebelum menganggap ambang masih sah.

deepface==0.0.100
easyocr==1.7.2

# DeepFace 0.0.100 berjalan di atas Keras 3 bawaan TensorFlow 2.21; paket
# tf-keras tidak diperlukan dan tidak terpasang di produksi.
tensorflow==2.21.0
numpy==2.5.0
opencv-python-headless==5.0.0.93

# Detektor utama pada rantai pelurusan wajah di verify_face.py. Dulu hanya ikut
# terbawa sebagai dependensi DeepFace; dicantumkan eksplisit karena sekarang
# kualitas seluruh sistem bergantung padanya.
retina-face==0.0.18

# Detektor cadangan bila retinaface gagal menemukan wajah.
mtcnn==1.0.0

# Dipakai easyocr; dicantumkan eksplisit agar versinya tidak berubah diam-diam.
torch==2.12.1
torchvision==0.27.1
